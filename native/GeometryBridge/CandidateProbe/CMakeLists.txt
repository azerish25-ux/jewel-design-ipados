cmake_minimum_required(VERSION 3.20)
project(JewelOcctCandidate LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT OCCT_SOURCE_DIR)
  message(FATAL_ERROR "Set OCCT_SOURCE_DIR to the pinned source checkout")
endif()
execute_process(COMMAND git -C "${OCCT_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE source_commit OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE git_status)
if(NOT git_status EQUAL 0 OR NOT source_commit STREQUAL "b8f597c677811d1f9f4d8a97f5ae2825c0353a42")
  message(FATAL_ERROR "OCCT candidate source commit does not match pin")
endif()
set(BUILD_LIBRARY_TYPE Static CACHE STRING "" FORCE)
set(BUILD_USE_VCPKG OFF CACHE BOOL "" FORCE)
set(BUILD_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_DOC_Overview OFF CACHE BOOL "" FORCE)
set(INSTALL_DIR_INCLUDE inc CACHE PATH "" FORCE)
foreach(module FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange Draw)
  set(BUILD_MODULE_${module} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_ADDITIONAL_TOOLKITS "TKPrim TKBO TKMesh TKBool" CACHE STRING "" FORCE)
foreach(dependency FREETYPE TK FREEIMAGE FFMPEG OPENVR RAPIDJSON DRACO TBB EIGEN XLIB)
  set(USE_${dependency} OFF CACHE BOOL "" FORCE)
endforeach()
# OCCT generates Standard_Version.hxx relative to its subproject binary directory,
# while collecting ordinary headers under the top-level binary directory.
include_directories("${CMAKE_CURRENT_BINARY_DIR}/occt/inc")
add_subdirectory("${OCCT_SOURCE_DIR}" occt)
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/occt/inc/Standard_Version.hxx")
  message(FATAL_ERROR "Pinned OCCT generated version header is missing from its subproject binary include directory")
endif()
set(ROTATION_REFERENCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../.local/reference/opennurbs6-rotation")
if(NOT EXISTS "${ROTATION_REFERENCE_DIR}/rotation-methods.inc")
  message(FATAL_ERROR "Run node scripts/prepare-rotation-reference.cjs before configuring native comparison tests")
endif()
file(SHA256 "${ROTATION_REFERENCE_DIR}/rotation-methods.inc" ROTATION_REFERENCE_HASH)
if(NOT ROTATION_REFERENCE_HASH STREQUAL "40fdcbece9ffcf20b50c02472869a337a96235f0327fd31b5301ed1751b09560")
  message(FATAL_ERROR "Prepared rotation comparison methods do not match their immutable pin")
endif()
set(PLANE_REFERENCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../.local/reference/opennurbs6-plane")
if(NOT EXISTS "${PLANE_REFERENCE_DIR}/plane-methods.inc")
  message(FATAL_ERROR "Run node scripts/prepare-plane-reference.cjs before configuring native comparison tests")
endif()
file(SHA256 "${PLANE_REFERENCE_DIR}/plane-methods.inc" PLANE_REFERENCE_HASH)
if(NOT PLANE_REFERENCE_HASH STREQUAL "a20609e59747a57a0b9124bdc7d5123e094a4126ddfa683b806b85afa7068af5")
  message(FATAL_ERROR "Prepared plane comparison methods do not match immutable pin")
endif()
add_library(jewel_occt_candidate_objects OBJECT probe.cpp candidate_abi_contracts.cpp candidate_curve_contracts.cpp candidate_abi.cpp rotation_reference.cpp plane_reference.cpp candidate_sweep_contracts.cpp candidate_paired_sweep_contracts.cpp candidate_paired_sweep.cpp candidate_paired_abi_contracts.cpp)
target_include_directories(jewel_occt_candidate_objects PRIVATE "${ROTATION_REFERENCE_DIR}" "${PLANE_REFERENCE_DIR}")
# Keep reference scalar operations separate on ARM64 and x64; no fused contraction.
if(MSVC)
  set_source_files_properties(plane_reference.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
else()
  set_source_files_properties(plane_reference.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
endif()
target_include_directories(jewel_occt_candidate_objects PRIVATE "${CMAKE_BINARY_DIR}/inc" "${CMAKE_CURRENT_BINARY_DIR}/occt/inc")
add_library(jewel_occt_candidate STATIC $<TARGET_OBJECTS:jewel_occt_candidate_objects>)
set_target_properties(jewel_occt_candidate PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(jewel_occt_candidate PUBLIC TKBool TKBO TKMesh TKPrim TKTopAlgo TKShHealing TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  add_executable(occt_candidate_test main.cpp)
  target_link_libraries(occt_candidate_test PRIVATE jewel_occt_candidate)
  enable_testing()
  add_test(NAME analytic_native_geometry COMMAND occt_candidate_test)
endif()

# Local Windows test module only; Apple apps keep static __Internal linking.
if(WIN32)
  add_library(jewel_occt_candidate_host SHARED $<TARGET_OBJECTS:jewel_occt_candidate_objects>)
  set_target_properties(jewel_occt_candidate_host PROPERTIES PREFIX "" WINDOWS_EXPORT_ALL_SYMBOLS ON LINKER_LANGUAGE CXX)
  target_link_libraries(jewel_occt_candidate_host PRIVATE jewel_occt_candidate)
endif()
